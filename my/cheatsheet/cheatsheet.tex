\section*{Setup \& notation}
Data $\{(\xn,y_n)\}_{n=1}^N$, $\xn\in\R^D$, $N$ = data size, $D$ = dimension.
Design matrix $\bm{X}\in\R^{N\times D}$ (rows $\xn^\top$), targets $\bm{y}\in\R^N$.
Offset: prepend a column of ones $\Rightarrow \tilde{\bm{X}}$, $\tilde{\w}=(w_0,\w)$.
Prediction (linear): $\hat{y}_n=\xn^\top\w$.

\section*{Loss functions}
\textbf{MSE} $\ \Lw=\frac{1}{2N}\sum_n (y_n-\xn^\top\w)^2=\frac{1}{2N}\|\bm{y}-\bm{X}\w\|^2$
\begin{itemize}
\item Convex, differentiable, closed form. Not robust to outliers ($e^2$ blows up).
\end{itemize}
\textbf{MAE} $\ \Lw=\frac{1}{N}\sum_n|y_n-\xn^\top\w|$
\begin{itemize}
\item Convex, robust to outliers, \emph{not} differentiable at 0 (subgradient).
\end{itemize}
\textbf{Huber} — quadratic for $|e|\le\delta$, linear beyond: convex, differentiable, robust.
\[\!\!\ell_\delta(e)=\begin{cases}\tfrac12 e^2 & |e|\le\delta\\ \delta(|e|-\tfrac12\delta)&|e|>\delta\end{cases}\]
\textbf{Trade-off:} statistical (robustness) vs.\ computational (convexity/smoothness).

\section*{Linear regression}
\textbf{Normal equations:} minimise MSE $\Rightarrow \bm{X}^\top\bm{X}\w=\bm{X}^\top\bm{y}$,
so $\w^\star=(\bm{X}^\top\bm{X})^{-1}\bm{X}^\top\bm{y}$ (if invertible).
\begin{itemize}
\item Cost $O(ND^2+D^3)$. Ill-conditioned / singular when $D>N$ or collinear features.
\item Gram matrix $\bm{X}^\top\bm{X}$ is PSD; invertible $\iff$ $\bm{X}$ has full column rank.
\end{itemize}
\textbf{Ridge:} $\min \frac{1}{2N}\|\bm{y}-\bm{X}\w\|^2+\lambda\|\w\|^2
\Rightarrow \w^\star=(\bm{X}^\top\bm{X}+\lambda' \bm{I})^{-1}\bm{X}^\top\bm{y}$.
Always invertible for $\lambda>0$. \emph{Mind the $2N$ convention: $\lambda'=2N\lambda$.}

\section*{Optimization}
\textbf{Gradient descent:} $\w^{(t+1)}=\w^{(t)}-\gamma\nabla\mathcal{L}(\w^{(t)})$.
For MSE: $\nabla\Lw=-\frac1N\bm{X}^\top(\bm{y}-\bm{X}\w)=-\frac1N\bm{X}^\top\bm{e}$.
\begin{itemize}
\item $\gamma$ too large $\to$ diverges; too small $\to$ slow. Step $O(ND)$.
\end{itemize}
\textbf{SGD:} pick $n$ at random, $\w \leftarrow \w-\gamma\nabla\ell_n(\w)$. Step $O(D)$.
Unbiased: $\mathbb{E}[\nabla\ell_n]=\nabla\mathcal{L}$. Needs decaying $\gamma_t$ to converge.
\textbf{Convexity:} $\mathcal{L}$ convex $\Rightarrow$ every local min is global.
$f$ convex $\iff \nabla^2 f\succeq 0$ $\iff f(\lambda a+(1-\lambda)b)\le\lambda f(a)+(1-\lambda)f(b)$.
Sum of convex is convex; convex $\circ$ affine is convex.
